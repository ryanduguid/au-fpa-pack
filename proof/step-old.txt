
      - name: Gitleaks over every pull request commit
        # On a pull request the action scans with --no-merges --first-parent,
        # so a commit that reaches the branch through a merge is never read,
        # nor is anything a merge commit itself adds. This scans every commit
        # between the base and the head with the gitleaks the action installed;
        # --diff-merges=remerge shows what a two-parent merge adds beyond its
        # parents. Git shows nothing for a merge of three or more parents, so
        # one of those fails the check, as does a range that holds no commit.
        if: github.event_name == 'pull_request'
        env:
          BASE_SHA: ${{ github.event.pull_request.base.sha }}
          HEAD_SHA: ${{ github.event.pull_request.head.sha }}
        run: |
          range="${BASE_SHA:?}..${HEAD_SHA:?}"
          # gitleaks exits 0 when its git log fails, so run the same log first.
          git log -p -U0 --diff-merges=remerge "$range" > /dev/null
          if [ -z "$(git rev-list -n 1 "$range")" ]; then
            echo "::error::No commit lies between the base and the head, so nothing would be scanned."
            exit 1
          fi
          octopus="$(git rev-list --min-parents=3 "$range")"
          if [ -n "$octopus" ]; then
            echo "::error::A merge of three or more parents cannot be scanned for what it adds: ${octopus//$'\n'/ }"
            exit 1
          fi
          gitleaks git --config .gitleaks.toml --redact --verbose --log-opts="--diff-merges=remerge $range"
